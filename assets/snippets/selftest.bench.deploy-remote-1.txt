    def _build_env_exports(self, api_mode=None, payload_size=None, extra_env=None):
        """构造远程环境变量 export 列表"""
        env_exports = []
        if api_mode:
            env_exports.append('export API_MODE={}'.format(api_mode))
        if payload_size:
            env_exports.append('export PAYLOAD_SIZE={}'.format(payload_size))
        if extra_env:
            for k, v in extra_env.items():
                env_exports.append('export {}={}'.format(k, v))
        return env_exports

    @staticmethod
    def _wrap_bash(cmd_parts, env_exports):
        """拼装 bash -c 包装（环境变量 + 命令）"""
        inner = '{} ; {}'.format(' ; '.join(env_exports), ' '.join(cmd_parts)) if env_exports else ' '.join(cmd_parts)
        return "bash -c '{}'".format(inner)
