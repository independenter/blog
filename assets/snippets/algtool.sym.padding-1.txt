    explicit CipherRef(const std::string& name) {
        c = EVP_CIPHER_fetch(nullptr, name.c_str(), nullptr);
    }
    ~CipherRef() { if (c) EVP_CIPHER_free(c); }
    CipherRef(const CipherRef&) = delete;
    CipherRef& operator=(const CipherRef&) = delete;
    operator const EVP_CIPHER*() const { return c; }
    explicit operator bool() const { return c != nullptr; }
};

bool SymmAlg::isAvailable(const std::string& cipherName) {
    CipherRef cf(cipherName);
    return static_cast<bool>(cf);
}
