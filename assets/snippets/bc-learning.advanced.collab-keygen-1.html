407 private static BigInteger computeChallenge(BigInteger[] R, BigInteger[] pAgg, byte[] message)
408 {
409 SM3Digest digest = new SM3Digest();
410
411 // R 的 x, y 坐标（32 字节大端）
412 byte[] rx = toBytes32(R[0]);
413 byte[] ry = toBytes32(R[1]);
414 digest.update(rx, 0, rx.length);
415 digest.update(ry, 0, ry.length);
416
417 // P_agg 的 x, y 坐标
418 byte[] px = toBytes32(pAgg[0]);
419 byte[] py = toBytes32(pAgg[1]);
420 digest.update(px, 0, px.length);
421 digest.update(py, 0, py.length);
422
423 // 消息
424 digest.update(message, 0, message.length);
425
426 byte[] hash = new byte[digest.getDigestSize()];
427 digest.doFinal(hash, 0);
