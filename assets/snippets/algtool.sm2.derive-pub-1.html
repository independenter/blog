261bool Sm2::verify(const Bytes& pub, const Bytes& msg, const Bytes& id,
262 bool hashMode, const Bytes& sig) {
263 if (pub.size() != 2 * FE || sig.size() != 2 * FE) return false;
264 Bytes eB = digest(pub, msg, id, hashMode);
265 if (eB.empty()) return false;
266 return verifyDigest(pub, eB, sig);
267}

269Bytes Sm2::encrypt(const Bytes& pub, const Bytes& plain, const Bytes& fixedR) {
270 if (pub.size() != 2 * FE || plain.empty()) return {};
271 EC_GROUP* g = newGroup(); if (!g) return {};
272 BN_CTX* ctx = BN_CTX_new();
273 const BIGNUM* n = EC_GROUP_get0_order(g);

275 Bytes out;
276 if (P) {
277 BIGNUM* r = BN_new();
278 bool fixed = (fixedR.size() == FE);
279 for (int attempt = 0; attempt < 64 && out.empty(); ++attempt) {
280 if (fixed) { BN_free(r); r = BN_bin2bn(fixedR.data(), FE, nullptr); if (BN_is_zero(r)) break; }
281 else { BN_rand_range(r, n); if (BN_is_zero(r)) continue; }
