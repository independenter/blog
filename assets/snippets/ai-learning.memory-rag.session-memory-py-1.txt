    def recall(self, query: str, top_k: int = 3) -> list[str]:
        """召回相关记忆"""
        if not self.memories:
            return []

        # 生成查询向量
        result = self.client.embed(query)
        query_embedding = result.get("embeddings", [[]])[0]

        if not query_embedding:
            return []

        # 计算相似度
        import numpy as np
        query_vec = np.array(query_embedding, dtype=np.float32)
        memory_vecs = np.array([m.embedding for m in self.memories if m.embedding], dtype=np.float32)

        if len(memory_vecs) == 0:
            return []
