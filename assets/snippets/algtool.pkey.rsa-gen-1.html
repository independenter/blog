138 Bytes out(len);
139 if (EVP_PKEY_decrypt(ctx, out.data(), &len, cipher.data(), cipher.size()) <= 0) {
140 logOp("dec", Rc::CryptoError, t.ms(), ERR_error_string(ERR_get_error(), nullptr));
141 EVP_PKEY_CTX_free(ctx); return {};
142 }
143 out.resize(len);
144 EVP_PKEY_CTX_free(ctx);
145 logOp("dec", Rc::Ok, t.ms(), "out=" + std::to_string(len) + "B");
146 return out;
147}

150 BIO* b = BIO_new(BIO_s_mem());
