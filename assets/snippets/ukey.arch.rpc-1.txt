bool UkeyRpcClient::ping(Json& data, std::string& err) {
    HttpResponse resp;
    std::string e;
    if (!http_.get(kUkeyRpcPath, resp, e)) {
        err = "Ukey 桥不可达 (" + host_ + ":" + std::to_string(port_) + "): " + e;
        return false;
    }
    bool ok = false;
    Json r = Json::parse(bodyText(resp.body), &ok);
    if (!ok || !r.at("ok").asBool()) {
        err = "Ukey 桥应答异常 (HTTP " + std::to_string(resp.status) + ")";
        return false;
    }
    data = r.at("d");
    return true;
}
