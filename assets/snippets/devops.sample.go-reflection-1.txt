func (info *ObjectInfo) processFields(v reflect.Value, t reflect.Type) {
	for i := 0; i < v.NumField(); i++ {
		field := t.Field(i)
		value := v.Field(i)
		
		// 如果是嵌套结构体，递归处理
		if field.Type.Kind() == reflect.Struct && field.Anonymous {
			info.processFields(value, field.Type)
			continue
		}
		
		if field.IsExported() { // 只处理可导出的字段
			info.Fields[field.Name] = value.Interface()
			info.FieldValue[field.Name] = value.Interface()
		}
	}
}
