    lay->addWidget(muted("文件存放在应用之下, 与容器平级; 容量受设备剩余空间限制。"));

    auto* tail = outBox(6);

                       const QString& extra = QString()) {
        tail->setPlainText(what + ": " + (ok ? "成功" : "失败 -- " + s2q(err)) +
                           (extra.isEmpty() ? QString() : "\n" + extra));
    };
