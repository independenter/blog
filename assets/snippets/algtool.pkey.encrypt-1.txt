static std::string toPem(EVP_PKEY* k, int priv) {
    BIO* b = BIO_new(BIO_s_mem());
    if (priv ? PEM_write_bio_PrivateKey(b, k, nullptr, nullptr, 0, nullptr, nullptr)
             : PEM_write_bio_PUBKEY(b, k)) {
        char* data = nullptr;
        long n = BIO_get_mem_data(b, &data);
        std::string s(data, data + (n > 0 ? n : 0));
        BIO_free(b);
        return s;
    }
    BIO_free(b);
