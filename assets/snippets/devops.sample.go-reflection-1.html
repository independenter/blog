52 if field.Type.Kind() == reflect.Struct && field.Anonymous {
53 info.processFields(value, field.Type)
54 continue
55 }
56
57 if field.IsExported() { // 只处理可导出的字段
58 info.Fields[field.Name] = value.Interface()

60 }
61 }
