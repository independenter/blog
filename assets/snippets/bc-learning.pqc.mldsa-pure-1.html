782 private static byte[] pack(int bits, int[] coeffs)
783 {
784 int nBytes = coeffs.length * bits / 8;
785 int[] out = new int[nBytes];
786 for (int i = 0; i < nBytes; i++)
787 {
788 int sum = 0;
789 for (int k = 0; k < 8; k++)
790 {
791 sum += ((coeffs[(8 * i + k) / bits] >>> ((8 * i + k) % bits)) & 1) << k;
792 }
793 out[i] = sum;
794 }
795 return bytesOf(out);
796 }
