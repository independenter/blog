42 int i = 0;
43 params[i++] = OSSL_PARAM_construct_utf8_string(OSSL_KDF_PARAM_DIGEST, mdBuf, 0);
44 params[i++] = OSSL_PARAM_construct_utf8_string(OSSL_KDF_PARAM_MODE, modeBuf, 0);
45 params[i++] = OSSL_PARAM_construct_octet_string(OSSL_KDF_PARAM_KEY, ikmC.data(), ikmC.size());
46 params[i++] = OSSL_PARAM_construct_octet_string(OSSL_KDF_PARAM_SALT,
47 saltC.empty() ? (void*)"" : saltC.data(), saltC.size());
48 params[i++] = OSSL_PARAM_construct_octet_string(OSSL_KDF_PARAM_INFO,
49 infoC.empty() ? (void*)"" : infoC.data(), infoC.size());
50 params[i] = OSSL_PARAM_construct_end();
51 if (EVP_KDF_derive(ctx, out.data(), oklen, params) <= 0) {
52 ALGLOG_ERROR(name_) << "HKDF failed: " << ERR_error_string(ERR_get_error(), nullptr);
53 out.clear();
54 } else {
55 ALGLOG_INFO(name_) << "hkdf ok md=" << md_ << " oklen=" << oklen << " cost=" << t.ms() << "ms";
56 }
57 EVP_KDF_CTX_free(ctx);
58 }
59 EVP_KDF_free(kdf);
60 return out;
61}
