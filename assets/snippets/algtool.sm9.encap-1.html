411 int rc = sm9_kem_encrypt(&impl_->enc_msk, id.data(), id.size(), klen, kbuf.data(), &C);
412 if (rc != 1) { logOp("encap", Rc::CryptoError, t.ms(), "id=" + id); return {}; }
413 // 以 65 字节非压缩坐标 (0x04||X||Y) 保存 C
414 outC.assign(65, 0);
415 sm9_z256_point_to_uncompressed_octets(&C, outC.data());
416 logOp("encap", Rc::Ok, t.ms(), "id=" + id + " klen=" + std::to_string(klen));
417 return kbuf;
418}
