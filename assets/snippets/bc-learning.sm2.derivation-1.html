43 int q = r / newR;
44 int tempT = t - q * newT;
45 t = newT;
46 newT = tempT;
47 int tempR = r - q * newR;
48 r = newR;
49 newR = tempR;
50 }
51 if (r > 1) throw new ArithmeticException("no inverse");
52 if (t < 0) t += mod;
53 return t;
54 }
