467 std::fprintf(stderr, "生成签名密钥对: %s\n", e2.c_str());
468 if (mgr.genKeyPair(true, pub, e2))
469 show("加密公钥 (令牌内生成)", toHex(pub));
470 else
471 std::fprintf(stderr, "生成加密密钥对: %s (数字信封与导入外部私钥需要它)\n", e2.c_str());
472 std::printf("接着签证书并写回: ukeyctl ca-issue --device --use-device-key --cn 你的名字\n");
473 return 0;
474}
