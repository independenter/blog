605 std::string body;
606 for (size_t i = text.find('\n', s) + 1; i < e2; i++)
607 if (text[i] != '\r' && text[i] != '\n') body += text[i];
608 b = b64Decode(body);
609 }
610 bool sign = !g_cli.flag("enc");
611 if (!mgr.importCert(sign, b, err)) die("写入容器失败: " + err);
612 std::printf("证书已写入容器 (%s 证书, %zu 字节)\n", sign ? "签名" : "加密", b.size());
613 std::printf("%s\n", SmallCa::describe(b).c_str());
614 return 0;
615}
