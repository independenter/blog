315 private static long[] keySchedule(byte[] key, boolean forEncryption)
316 {
317 long[] rk = expandKey(key);
318 if (forEncryption)
319 {
320 return rk;
321 }
322 long[] reversed = new long[32];
323 for (int i = 0; i < 32; i++)
324 {
325 reversed[i] = rk[31 - i];
326 }
327 return reversed;
328 }
