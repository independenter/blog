113class Logger {
114 constructor(dir, level) {
115 this.dir = abs(dir || 'logs');
116 this.level = level || 'INFO';
117 fs.mkdirSync(this.dir, { recursive: true });
118 this.stream = fs.createWriteStream(path.join(this.dir, 'ukeysite.log'), { flags: 'a' });
119 }
120 write(level, msg) {
121 const line = new Date().toISOString() + ' [' + level + '] ' + msg;
122 this.stream.write(line + '\n');
123 if (level !== 'TRACE') console.log(line);
124 }
125 info(m) { this.write('INFO', m); }
126 warn(m) { this.write('WARN', m); }
127 error(m) { this.write('ERROR', m); }
128}

130module.exports = { loadConfig, parseArgs, Logger, serveSite, startSite, abs, ROOT };
