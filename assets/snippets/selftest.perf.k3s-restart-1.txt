    def run(cmd, desc="", timeout=60):
        if desc:
            print("\n>> {}".format(desc))
        try:
            _, out, err = r.exec(cmd, timeout=timeout)
            out = out.strip()
            err = err.strip()
            if err:
                print("[stderr] {}".format(err[:300]))
            if out:
                print(out[:5000])
            return out
        except Exception as e:
            print("[ERROR] {}".format(e))
            return ""
