    bool useUkey = g_cli.flag("ukey");
    std::string keyPath;
    std::unique_ptr<SecClient> pc = makeClient(useUkey, keyPath);
    SecClient& cli2 = *pc;
    if (g_cli.flag("insecure")) cli2.setInsecureNoCertCheck(true);
    std::string op = g_cli.get("op");
    if (op.empty()) die("请指定 --op NAME (例如 ukey.info / echo / ca.list)");

    std::printf("安全通道已建立: sid=%s  服务端公钥=%s\n", cli2.sid().c_str(),
                toHex(cli2.serverPub()).c_str());

    std::string result;
