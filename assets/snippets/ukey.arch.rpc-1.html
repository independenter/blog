111bool UkeyRpcClient::ping(Json& data, std::string& err) {
112 HttpResponse resp;
113 std::string e;
114 if (!http_.get(kUkeyRpcPath, resp, e)) {
115 err = "Ukey 桥不可达 (" + host_ + ":" + std::to_string(port_) + "): " + e;
116 return false;
117 }
118 bool ok = false;
119 Json r = Json::parse(bodyText(resp.body), &ok);
120 if (!ok || !r.at("ok").asBool()) {
121 err = "Ukey 桥应答异常 (HTTP " + std::to_string(resp.status) + ")";
122 return false;
123 }
124 data = r.at("d");
125 return true;
126}
