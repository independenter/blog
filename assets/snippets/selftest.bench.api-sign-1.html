61def parse_size(value):
62 """解析人类可读的大小为字节数"""
63 if value is None:
64 return None
65 s = str(value).strip().upper().replace("B", "")
66 m = re.match(r"^([0-9]+(?:\.[0-9]+)?)\s*([KMG]?)$", s)
67 if not m:
68 raise ValueError("无法解析 PAYLOAD_SIZE: {!r}".format(value))
69 num = float(m.group(1))
70 mult = {"": 1, "K": 1024, "M": 1024 ** 2, "G": 1024 ** 3}[m.group(2)]
71 return int(num * mult)
