    def run(cmd, desc=""):
        if desc:
            print("\n>> {}".format(desc))
        print("$ {}".format(cmd))
        try:
            _, out, err = r.exec(cmd, timeout=30)
            out = out.strip()
            if err.strip():
                print("[stderr] {}".format(err.strip()[:200]))
            if out:
                print(out[:3000])
            return out
        except Exception as e:
            print("[ERROR] {}".format(e))
            return ""
