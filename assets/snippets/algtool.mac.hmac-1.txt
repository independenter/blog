    EVP_MAC* m = EVP_MAC_fetch(nullptr, "CMAC", nullptr);
    if (!m) { ALGLOG_ERROR(name_) << "CMAC provider unavailable"; return {}; }
    EVP_MAC_CTX* ctx = EVP_MAC_CTX_new(m);
    Bytes out;
    if (ctx) {
        OSSL_PARAM params[2];
        char cipherBuf[64];
        // 传入分组密码名 (如 "SM4-CBC"/"AES-128-CBC"), CMAC 会取其分组与密钥
        std::strncpy(cipherBuf, alg_.c_str(), sizeof(cipherBuf) - 1);
        cipherBuf[sizeof(cipherBuf) - 1] = '\0';
        params[0] = OSSL_PARAM_construct_utf8_string("cipher", cipherBuf, 0);
        params[1] = OSSL_PARAM_construct_end();
        if (EVP_MAC_init(ctx, key.data(), key.size(), params) == 1 &&
            EVP_MAC_update(ctx, data.data(), data.size()) == 1) {
            unsigned char tag[EVP_MAX_MD_SIZE]; size_t tagl = 0;
            if (EVP_MAC_final(ctx, tag, &tagl, sizeof(tag)) == 1) out.assign(tag, tag + tagl);
        }
