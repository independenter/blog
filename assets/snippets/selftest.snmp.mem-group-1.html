88def run_snmpget(oid):
89 """执行 snmpget 命令"""
90 cmd = [
91 "snmpget", "-v", SNMP_VERSION, "-c", SNMP_COMMUNITY,
92 SNMP_HOST, oid
93 ]
94 try:
95 result = subprocess.run(cmd, capture_output=True, text=True, timeout=10)
96 if result.returncode == 0:
97 # 解析输出: SNMPv2-MIB::sysDescr.0 = STRING: Linux ...
98 output = result.stdout.strip()
99 if "=" in output:
100 value = output.split("=", 1)[1].strip()
101 return {"status": "OK", "value": value}
102 return {"status": "FAIL", "error": result.stderr.strip() or "No response"}
103 except subprocess.TimeoutExpired:
104 return {"status": "TIMEOUT", "error": "Command timed out"}
105 except Exception as e:
106 return {"status": "ERROR", "error": str(e)}
