function base128(v, out) {
  const stack = [];
  let n = v;
  stack.unshift(n & 0x7f);
  n = Math.floor(n / 128);
  while (n > 0) { stack.unshift((n & 0x7f) | 0x80); n = Math.floor(n / 128); }
  for (const x of stack) out.push(x);
}

function oid(dotted) {
  const parts = String(dotted).split('.').map(Number);
  const bs = [];
  base128(parts[0] * 40 + parts[1], bs);
  for (let i = 2; i < parts.length; i++) base128(parts[i], bs);
  return tagBytes(0x06, new Uint8Array(bs));
}
