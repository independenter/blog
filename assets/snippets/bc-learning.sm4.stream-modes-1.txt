    private static long[] keySchedule(byte[] key, boolean forEncryption)
    {
        long[] rk = expandKey(key);
        if (forEncryption)
        {
            return rk;
        }
        long[] reversed = new long[32];
        for (int i = 0; i < 32; i++)
        {
            reversed[i] = rk[31 - i];
        }
        return reversed;
    }
