        /// <summary>获取所有工具分类</summary>
        public string[] GetCategories()
        {
            var categories = new List<string>();
            foreach (var tool in _tools)
            {
                if (!categories.Contains(tool.Category))
                    categories.Add(tool.Category);
            }
            return categories.ToArray();
        }
