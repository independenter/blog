static bool sm3Hash(const uint8_t* d, size_t n, uint8_t out[32]) {
    SM3_CTX ctx;
    sm3_init(&ctx);
    sm3_update(&ctx, d, n);
    sm3_finish(&ctx, out);
    return true;
}
