bool CertEnv::verifyRequest(const Bytes& req) {
    Chrono t;
    bool ok = !req.empty() && x509_req_verify(req.data(), req.size(),
                                              kSm2CertId, kSm2CertIdLen) == 1;
    return logOp("verifyRequest", ok ? Rc::Ok : Rc::VerifyFailed, t.ms(),
                 ok ? "请求自验通过" : "请求签名无效") == Rc::Ok;
}
