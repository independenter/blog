    @GetMapping("/memories")
    public Map<String, Object> memories() {
        return Map.of("memories", memory.all());
    }

    @GetMapping("/memories/recall")
    public Map<String, Object> recall(@RequestParam(defaultValue = "") String query,
                                      @RequestParam(name = "top_k", defaultValue = "3") int topK) {
        List<String> results = query.isBlank() ? List.of() : memory.recall(query, topK);
        return Map.of("memories", results, "query", query);
    }
