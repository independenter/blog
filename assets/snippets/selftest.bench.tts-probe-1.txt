    def __init__(self, host=None, port=None, username=None, password=None,
                 key_file=None, base_dir=None, targets=None, threads=None,
                 duration_s=None, data_len=None, timeout_extra=None):
        cfg = get_config('tts') or {}
        ssh_cfg = get_config('ssh') or {}

        self.host = host or cfg.get('ip')
        self.port = int(port or cfg.get('port') or ssh_cfg.get('port') or 22)
        self.username = username or cfg.get('username')
        self.password = ((cfg.get('password') or '') if password is None else password)
        self.key_file = key_file if key_file is not None else cfg.get('key_file')
        self.base_dir = (base_dir or cfg.get('base_dir') or '').rstrip('/')
        self.targets = list(targets if targets is not None else (cfg.get('targets') or []))
        self.threads = int(threads or cfg.get('threads') or 30)
        self.duration_s = int(duration_s or cfg.get('duration_s') or 30)
        self.data_len = int(data_len or cfg.get('data_len') or 32)
        self.timeout_extra = int(timeout_extra or cfg.get('timeout_extra') or 60)
        # SM2 每线程生成密钥，设备偶发 SDF_GenerateKeyPair_ECC 失败：默认重试 1 次
        self.retries = int(cfg['retry']) if cfg.get('retry') is not None else 1
        self.retry_delay_s = (int(cfg['retry_delay_s'])
                              if cfg.get('retry_delay_s') is not None else 3)
        self._ssh = None
