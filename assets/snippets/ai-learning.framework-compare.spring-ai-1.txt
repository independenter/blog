    @PostMapping("/reset")
    public Map<String, Object> reset(@RequestBody ResetRequest req) {
        String session = req.sessionIdOrDefault();
        chatMemory.clear(session);
        Map<String, Object> out = new HashMap<>();
        out.put("session_id", session);
        out.put("message", "会话已重置");
        return out;
    }
