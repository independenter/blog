    alg::Bytes plainBytes() {
        QString h = edPtHex_->text().trimmed();
        if (!h.isEmpty()) { try { return alg::fromHex(h.toStdString()); } catch (...) {} }
        return algui::textBytes(ptText_);
    }

    alg::Bytes idBytes() {
        alg::Bytes id = algui::hexBytes(edId_);
        if (id.empty()) id = alg::bytesFromString("1234567812345678");
        return id;
    }
