275 void doEnc(QLineEdit* randField) {
276 alg::Bytes pub = pubOrDerive();
277 if (pub.empty()) { status("加密失败: 无有效公钥"); return; }
278 alg::Bytes r = randField ? algui::hexBytes(randField) : alg::Bytes{};
279 if (randField && r.size() != 32) { status("加密失败: 固定随机数需 32 字节"); return; }
280 alg::Bytes pt = plainBytes();
281 if (pt.empty()) { status("加密失败: 明文为空"); return; }
282 prov::PkeyProvider* p = provider();
283 if (!p) { status("加密失败: 无可用的执行后端 (点\"⟳ 探测\"或到\"管理…\"配置 Agent)"); return; }
284 prov::PkeyRequest req;
285 req.op = 0; req.pub = pub; req.plain = pt; req.fixedRand = r;
286 prov::PkeyResult res = p->run(req);
287 if (!res.ok || res.out.empty()) { status(QString("加密失败: ") + (res.error.isEmpty() ? QStringLiteral("公钥非曲线有效点") : res.error)); return; }
288 edCt_->setPlainText(QString::fromStdString(alg::toHex(res.out)));
289 QString note = randField ? QStringLiteral("固定随机数加密完成 ✓") : QStringLiteral("加密完成 ✓");
290 if (randField && !bar_->currentIsSoftware()) note += QStringLiteral(" (硬件忽略固定随机数, r 由设备产生)");
291 status(QString("[%1] ").arg(p->name()) + note);
292}
