    void refreshUi() {
        prov::SymmProvider* p = provider();
        const bool have = (p != nullptr);
        const QString m = mode();
        const int ksz = have ? p->keySize(base_, m) : 0;
        const bool noIv = (m == "ECB");
        const int vsz = (have && !noIv) ? p->ivSize(base_, m) : 0;

        const bool blockMode = (m == "ECB" || m == "CBC");
        padCb_->setEnabled(have && blockMode);
        macBtn_->setEnabled(have && p->supportsMac(base_));
        encBtn_->setEnabled(have);
        decBtn_->setEnabled(have);
        ivEd_->setEnabled(have && !noIv);
