493bool CertEnv::verifyByCa(const Bytes& der, const Bytes& caDer) {
494 Chrono t;
495 bool ok = !der.empty() && !caDer.empty() &&
496 x509_cert_verify_by_ca_cert(der.data(), der.size(), caDer.data(), caDer.size(),
497 kSm2CertId, kSm2CertIdLen) == 1;
498 return logOp("verifyByCa", ok ? Rc::Ok : Rc::VerifyFailed, t.ms(),
499 ok ? "证书链验证通过" : "CA 验证失败 (签名/ issuer 不匹配)") == Rc::Ok;
500}
501
502bool CertEnv::verifySelfSigned(const Bytes& der) {
503 Chrono t;
504 const Bytes& data = der.empty() ? impl_->cert : der;
505 // 自签 = 自发行(is_self_issued) 且用自己的主体公钥能验过签名 (以自身为 CA 验证)
506 bool ok = !data.empty() &&
507 x509_cert_is_self_issued(data.data(), data.size()) == 1 &&
508 x509_cert_verify_by_ca_cert(data.data(), data.size(),
509 data.data(), data.size(),
510 kSm2CertId, kSm2CertIdLen) == 1;
511 return logOp("verifySelfSigned", ok ? Rc::Ok : Rc::VerifyFailed, t.ms(),
512 ok ? "自签验证通过" : "自签验证失败") == Rc::Ok;
513}
