30 if (EVP_DigestInit_ex(ctx, md, nullptr) == 1 &&
31 EVP_DigestUpdate(ctx, msg.data(), msg.size()) == 1 &&
32 EVP_DigestFinal_ex(ctx, out.data(), &len) == 1) {
33 out.resize(len);
34 hex = toHex(out);
35 if (outRaw) *outRaw = out;
36 logOp("hash:" + name_, Rc::Ok, t.ms(), "in=" + std::to_string(msg.size()) + "B digest=" + hex);
37 } else {
38 logOp("hash:" + name_, Rc::CryptoError, t.ms(), ERR_error_string(ERR_get_error(), nullptr));
39 hex.clear();
40 }
41
42 EVP_MD_CTX_free(ctx);
43 return hex;
44}
