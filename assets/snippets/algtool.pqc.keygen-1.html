359 return logOp("verify", Rc::VerifyFailed, t.ms(), "签名解析失败") == Rc::Ok;
360 }
361 int rc = 1;
362 if (!msg.empty()) rc = lms_verify_update(&ctx, msg.data(), msg.size());
363 if (rc == 1) rc = lms_verify_finish(&ctx);
364 bool ok = (rc == 1);
365 return logOp("verify", ok ? Rc::Ok : Rc::VerifyFailed, t.ms(),
366 ok ? "签名有效" : "签名无效") == Rc::Ok;
367}
