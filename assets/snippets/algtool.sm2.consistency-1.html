200 bool ok = inv && !BN_is_zero(s);
201 BN_free(d1); BN_free(inv); BN_free(rd); BN_free(krd);
202 if (!ok) { if (fixed) break; continue; }
203
204 Bytes br = bn_pad(r), bs = bn_pad(s);
205 out.insert(out.end(), br.begin(), br.end());
206 out.insert(out.end(), bs.begin(), bs.end());
207 done = true;
208 }
209 BN_free(e); BN_free(k); BN_free(r); BN_free(s);
210 BN_free(d); BN_CTX_free(ctx); EC_GROUP_free(g);
211 if (out.empty()) ALGLOG_ERROR("SM2") << "sign failed";
212 return out;
213}
