    public Map<String, Object> getStatus() {
        Map<String, Object> out = new LinkedHashMap<>();
        out.put("provider", currentName);
        out.put("provider_type", providerTypes.getOrDefault(currentName, ""));
        out.put("model", currentModel);
        out.put("embed_model", embedModel);
        List<Map<String, Object>> ps = new ArrayList<>();
        providers.forEach((n, m) -> ps.add(Map.of("name", n, "type", providerTypes.getOrDefault(n, ""))));
        out.put("providers", ps);
        return out;
    }
