257 return 0
258 fi

260 # ufw (Ubuntu)
261 if command -v ufw >/dev/null 2>&1; then

263 log_ok "ufw: 已开放 UDP ${SNMP_PORT} 端口"
264 return 0
265 fi
266
267 log_warn "未检测到防火墙工具，请手动开放 UDP ${SNMP_PORT} 端口"
268}
