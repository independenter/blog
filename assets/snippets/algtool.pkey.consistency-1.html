117 Bytes out(len);
118 if (EVP_PKEY_encrypt(ctx, out.data(), &len, plain.data(), plain.size()) <= 0) {
119 logOp("enc", Rc::CryptoError, t.ms(), ERR_error_string(ERR_get_error(), nullptr));
120 EVP_PKEY_CTX_free(ctx); return {};
121 }
122 out.resize(len);
123 EVP_PKEY_CTX_free(ctx);
124 logOp("enc", Rc::Ok, t.ms(), "out=" + std::to_string(len) + "B");
125 return out;
126}

129 Chrono t;
130 if (kind_ != Kind::RSA || !pkey_) { logOp("dec", Rc::InvalidArg, t.ms(), "only RSA supports dec"); return {}; }
131 EVP_PKEY_CTX* ctx = EVP_PKEY_CTX_new(pkey_, nullptr);
132 if (!ctx || EVP_PKEY_decrypt_init(ctx) <= 0) {
133 logOp("dec", Rc::CryptoError, t.ms(), ERR_error_string(ERR_get_error(), nullptr));
134 if (ctx) EVP_PKEY_CTX_free(ctx); return {};
135 }
136 size_t len = 0;
137 EVP_PKEY_decrypt(ctx, nullptr, &len, cipher.data(), cipher.size());
