    def resume_after_approval(self, session_id: str = "default") -> Generator[dict, None, None]:
        pending = self._pending_approvals.pop(session_id, None)
        if not pending:
            yield {"type": "error", "content": "没有待审批的操作"}
            return

        tool_name = pending["tool_name"]
        tool_args = pending["tool_args"]
        messages = pending["messages"]
        response_text = pending["response_text"]
        start_iteration = pending["iteration"] + 1
        invalid_tool_retries = pending["invalid_tool_retries"]

        self.approved_tools.add(tool_name)
        tool = self._get_tool(tool_name)
        if tool is None:
            yield {"type": "error", "content": f"工具 {tool_name} 不存在"}
