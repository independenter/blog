15// 把当前列表控件重新填成 items
16//
17// 可编辑下拉框清空后 currentIndex 变 -1, 此时 currentText() 是空串 —— 若不
18// 重新指定当前项, 用户点"打开应用"就会拿空名字去 SKF_OpenApplication, 报
19// "应用不存在" (刚创建完就打不开, 很莫名)。这里保证: 优先保留原选择,
20// 原选择已不存在或本来为空时, 落到列表第一项。
21void fillList(QComboBox* box, const std::vector<std::string>& items) {
22 QString cur = box->currentText();
23 box->clear();
24 for (const auto& s : items) box->addItem(s2q(s));
25 if (box->count() == 0) {
26 box->setPlaceholderText("(无)");
27 return;
28 }
