    public Map<String, Object> page(int pageNo, int pageSize) {
        int total = chunkIndex.size();
        int start = Math.max(0, (pageNo - 1) * pageSize);
        int end = Math.min(total, start + pageSize);
        List<Map<String, Object>> slice = start < end ? chunkIndex.subList(start, end) : List.of();
        Map<String, Object> out = new HashMap<>();
        out.put("chunks", slice);
        out.put("total", total);
        out.put("page", pageNo);
        out.put("page_size", pageSize);
        out.put("total_pages", total == 0 ? 0 : (total + pageSize - 1) / pageSize);
        return out;
    }
