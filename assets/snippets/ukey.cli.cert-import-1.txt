        ProviderUrl u;

            die("ukey 提供地址必须是 http(s) URL: " + setv);
        g_cfg.set("service", "provider_url", setv);

                    setv.c_str(), g_cfg.path().c_str());
        return 0;
    }

    ProviderUrl u;
    if (!splitProviderUrl(url, u)) die("地址无法解析: " + url);
    if (u.scheme != "http") {
        std::printf("  探测跳过: 内置 HTTP 客户端只走明文, %s 地址请手工验证\n", u.scheme.c_str());
