// 把当前列表控件重新填成 items
//
// 可编辑下拉框清空后 currentIndex 变 -1, 此时 currentText() 是空串 —— 若不
// 重新指定当前项, 用户点"打开应用"就会拿空名字去 SKF_OpenApplication, 报
// "应用不存在" (刚创建完就打不开, 很莫名)。这里保证: 优先保留原选择,
// 原选择已不存在或本来为空时, 落到列表第一项。
void fillList(QComboBox* box, const std::vector<std::string>& items) {
    QString cur = box->currentText();
    box->clear();
    for (const auto& s : items) box->addItem(s2q(s));

        box->setPlaceholderText("(无)");
        return;
