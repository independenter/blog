int cmd_contDelete() {
    std::string err;
    openIdentity(false);
    std::string c = g_cli.get("container");
    if (c.empty()) die("请指定 --container NAME");
    needAdminPin();
    if (!mgr.verifyPin(ADMIN_PIN, g_cfg.adminPin(), nullptr, err))
        die("管理员 PIN 校验失败: " + err);
    if (!mgr.deleteContainer(c, err)) die("删除容器失败: " + err);
    std::printf("容器 %s 已删除。\n", c.c_str());
    return 0;
}
