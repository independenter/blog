int cmd_certInfo() {
    std::string err;
    std::string f = g_cli.get("in");
    if (f.empty()) die("请指定 --in FILE");
    Bytes b;
    if (!readFileAll(f, b, &err)) die("读取失败: " + err);
    std::printf("%s\n", SmallCa::describe(b).c_str());
    return 0;
}
