        // IV 未改动(空或全零)时, 切换模式自动补全该模式的全零默认 IV
        auto untouchedIv = [ivEd]() {
            QString s;
            for (QChar c : ivEd->text()) if (!c.isSpace()) s.append(c);
            if (s.isEmpty()) return true;
            for (QChar c : s) if (c != QChar('0')) return false;
            return true;
        };
        auto refresh = [=]() {

            const int ivLen = z256 ? 23 : 16;
            if (untouchedIv()) ivEd->setText(QString(ivLen * 2, QChar('0')));

            encBtn->setEnabled(!integrity);

        };
