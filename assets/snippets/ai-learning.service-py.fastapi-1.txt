@app.post("/api/remember")
def remember(req: dict):
    fact = req.get("fact", "")
    if fact:
        agent_service.remember(fact)
    return {"status": "ok", "fact": fact}


@app.get("/api/memories")
def memories():
    return {"memories": agent_service.get_memories()}


@app.get("/api/memories/recall")
def recall_memories(query: str = "", top_k: int = 3):
    if not query:
        return {"memories": [], "query": ""}
    results = agent_service.recall_memories(query, top_k)
    return {"memories": results, "query": query}
