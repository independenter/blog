349 private static byte[] kdf(byte[] z, int klen)
350 {
351 int blocks = Calc.ceilDiv(klen, 32);
352 byte[] stream = new byte[0];
353 for (int ct = 1; ct <= blocks; ct++)
354 {
355 // 把 32-bit 计数器 ct 摊成 4 个字节，高位在前（大端）：i = 0 取第 31..24 位
356 byte[] counterBytes = new byte[4];
357 for (int i = 0; i < 4; i++)
358 {
359 counterBytes[i] = (byte)(ct >>> (24 - 8 * i));
360 }
361 stream = Calc.concat(stream, Sm3.sm3(Calc.concat(z, counterBytes)));
362 }
363 return Arrays.copyOf(stream, klen);
364 }
