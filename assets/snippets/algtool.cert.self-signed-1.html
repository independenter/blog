30std::string utcText(time_t t) {
31 long long days = static_cast<long long>(t) / 86400;
32 int secs = static_cast<int>(static_cast<long long>(t) % 86400);
33 if (secs < 0) { secs += 86400; --days; }
34 // Howard Hinnant civil_from_days
35 days += 719468;
36 long long era = (days >= 0 ? days : days - 146096) / 146097;
37 long long doe = days - era * 146097;
38 long long yoe = (doe - doe / 1460 + doe / 36524 - doe / 146096) / 365;
39 long long y = yoe + era * 400;
40 long long doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
41 long long mp = (5 * doy + 2) / 153;
42 long long d = doy - (153 * mp + 2) / 5 + 1;
43 long long m = mp + (mp < 10 ? 3 : -9);
44 if (m <= 2) ++y;
45 char buf[32];
46 std::snprintf(buf, sizeof(buf), "%04lld-%02lld-%02lld %02d:%02d:%02d UTC",
47 y, m, d, secs / 3600, (secs % 3600) / 60, secs % 60);
48 return buf;
49}
