104Bytes kdf(const uint8_t* xb, const uint8_t* yb, size_t len) {
105 Bytes t; uint32_t ctr = 1;
106 while (t.size() < len) {
107 uint8_t cb[4] = {uint8_t(ctr >> 24), uint8_t(ctr >> 16), uint8_t(ctr >> 8), uint8_t(ctr)};
108 EVP_MD_CTX* mc = EVP_MD_CTX_new();
109 EVP_DigestInit(mc, EVP_sm3());
110 EVP_DigestUpdate(mc, xb, FE); EVP_DigestUpdate(mc, yb, FE); EVP_DigestUpdate(mc, cb, 4);
111 uint8_t dig[32]; unsigned dl = 0; EVP_DigestFinal(mc, dig, &dl);
112 EVP_MD_CTX_free(mc);
113 t.insert(t.end(), dig, dig + dl); ++ctr;
114 }
115 t.resize(len);
116 return t;
117}
