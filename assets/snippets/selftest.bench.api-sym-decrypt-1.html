90def size_label(nbytes):
91 """字节数 → 可读标签"""
92 for unit, mult in [("M", 1024 ** 2), ("K", 1024)]:
93 if nbytes % mult == 0 and nbytes >= mult:
94 return "{}{}".format(nbytes // mult, unit)
95 return str(nbytes)
