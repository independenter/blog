769 bool enc = !g_cli.flag("dec");
770 uint32_t alg = (uint32_t)g_cli.getInt("alg-int", SGD_SM4_CBC);
771 std::string a = lower(g_cli.get("alg"));
772 if (a == "sm4-ecb") alg = SGD_SM4_ECB;
773 Bytes out;
774 if (!mgr.symmCrypt(key, alg, iv, data, out, enc, err)) die("对称运算失败: " + err);
775 if (!writeOut(out, enc ? "密文" : "明文")) std::printf("%s\n", toHex(out).c_str());
776 return 0;
777}
