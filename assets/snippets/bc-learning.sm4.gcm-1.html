548 private static byte[] cfbEncrypt(long[] roundKeys, byte[] iv, byte[] data)
549 {
550 List<byte[]> parts = Calc.chunks(data, BS);
551 // prev 是公式里的反馈块 C_{i−1}，第一圈取 IV
552 byte[] prev = iv;
553 byte[] out = new byte[0];
554 for (int i = 0; i < parts.size(); i++)
555 {
556 // Cᵢ = Pᵢ ⊕ E_K(C_{i−1})：加密器的输入是上一块密文，不是计数器、也不是自己的输出
557 byte[] c = Calc.xor(sm4Block(roundKeys, prev), parts.get(i));
558 out = Calc.concat(out, c);
559 prev = Arrays.copyOf(c, BS);
560 }
561 return out;
562 }
