50 Bytes ks(nwords * 4);
51 for (size_t w = 0; w < nwords; ++w) {
52 ZUC_UINT32 word = zuc256_generate_keyword(&st);
53 ks[w * 4 + 0] = static_cast<uint8_t>(word & 0xFF);
54 ks[w * 4 + 1] = static_cast<uint8_t>((word >> 8) & 0xFF);
55 ks[w * 4 + 2] = static_cast<uint8_t>((word >> 16) & 0xFF);
56 ks[w * 4 + 3] = static_cast<uint8_t>((word >> 24) & 0xFF);
57 }
58 Bytes out(data.size());
59 for (size_t i = 0; i < data.size(); ++i) out[i] = static_cast<uint8_t>(data[i] ^ ks[i]);
60 logOp("crypt256", Rc::Ok, t.ms(), "len=" + std::to_string(out.size()));
61 return out;
62}
