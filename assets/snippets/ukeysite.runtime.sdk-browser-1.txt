function mod(a, m) {
  const r = a % m;
  return r >= 0n ? r : r + m;
}

function intToBytes(v, nbytes) {
  let hex = v.toString(16);
  if (hex.length > nbytes * 2) throw new UkeyError('整数超过 ' + nbytes + ' 字节');
  return fromHex(hex.padStart(nbytes * 2, '0'));
}
