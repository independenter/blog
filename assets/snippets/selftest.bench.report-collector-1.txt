    def mark_vsm_result(self, vsm_name, exit_code, records_before):
        """记录单个 VSM 测试结果（exit_code + 是否有数据）
        records_before: 调用 parse_stats_file 前的 len(report._records)
        """
        has_data = len(self._records) > records_before
        passed = self._metadata.setdefault('vsms_passed', [])
        failed = self._metadata.setdefault('vsms_failed', [])
        if exit_code == 0 and has_data:
            if vsm_name not in passed:
                passed.append(vsm_name)
        else:
            if vsm_name not in failed:
                failed.append(vsm_name)
