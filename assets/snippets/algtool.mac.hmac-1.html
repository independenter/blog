38 std::strncpy(cipherBuf, alg_.c_str(), sizeof(cipherBuf) - 1);
39 cipherBuf[sizeof(cipherBuf) - 1] = '\0';
40 params[0] = OSSL_PARAM_construct_utf8_string("cipher", cipherBuf, 0);
41 params[1] = OSSL_PARAM_construct_end();
42 if (EVP_MAC_init(ctx, key.data(), key.size(), params) == 1 &&
43 EVP_MAC_update(ctx, data.data(), data.size()) == 1) {

45 if (EVP_MAC_final(ctx, tag, &tagl, sizeof(tag)) == 1) out.assign(tag, tag + tagl);
