78function base128(v, out) {
79 const stack = [];
80 let n = v;
81 stack.unshift(n & 0x7f);
82 n = Math.floor(n / 128);
83 while (n > 0) { stack.unshift((n & 0x7f) | 0x80); n = Math.floor(n / 128); }
84 for (const x of stack) out.push(x);
85}
