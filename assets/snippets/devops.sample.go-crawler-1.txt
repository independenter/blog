func (c *WebCrawler) extractTitle(n *html.Node) string {
	if n.Type == html.ElementNode && n.Data == "title" {
		if n.FirstChild != nil {
			return n.FirstChild.Data
		}
		return ""
	}

	for child := n.FirstChild; child != nil; child = child.NextSibling {
		if title := c.extractTitle(child); title != "" {
			return title
		}
	}

	return ""
}
