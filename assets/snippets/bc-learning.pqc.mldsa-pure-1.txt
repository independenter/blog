    private static byte[] pack(int bits, int[] coeffs)
    {
        int nBytes = coeffs.length * bits / 8;
        int[] out = new int[nBytes];
        for (int i = 0; i < nBytes; i++)
        {
            int sum = 0;
            for (int k = 0; k < 8; k++)
            {
                sum += ((coeffs[(8 * i + k) / bits] >>> ((8 * i + k) % bits)) & 1) << k;
            }
            out[i] = sum;
        }
        return bytesOf(out);
    }
