UK_EXPORT ULONG DEVAPI SKF_EnumContainer(HAPPLICATION hApplication, LPSTR szContainerName,
                                         ULONG* pulSize) {
    std::vector<std::string> names;
    ULONG rc = E().enumConts(hApplication, names);
    if (rc != SAR_OK) return rc;
    return fillNameList(names, szContainerName, pulSize);
}
