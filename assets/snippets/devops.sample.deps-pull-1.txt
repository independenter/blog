def get_script_and_project_dirs():
    """获取脚本目录和项目根目录"""
    script_dir = Path(__file__).resolve().parent
    project_dir = script_dir.parent.parent
    return script_dir, project_dir


def parse_arguments():
    """解析命令行参数"""
    logger = logging.getLogger(__name__)
    parser = argparse.ArgumentParser(description='下载Maven/Gradle依赖')
    parser.add_argument('--force', action='store_true', 
                        help='强制重新下载所有依赖，删除现有lib目录')
    parser.add_argument('--gradle', action='store_true',
                        help='使用Gradle而不是Maven下载依赖')
    
    args = parser.parse_args()
    logger.info(f"命令行参数: force={args.force}, gradle={args.gradle}")
    return args
